
Install Nginix

# 1. Update your package list
sudo apt update

# 2. Install Nginx
sudo apt install nginx -y

# 3. Start Nginx
sudo systemctl start nginx

# 4. Enable Nginx to auto-start on boot
sudo systemctl enable nginx

# 5. Check status
sudo systemctl status nginx
